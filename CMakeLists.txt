cmake_minimum_required(VERSION 3.20)
project(licensebox_fortran VERSION 0.1.0 LANGUAGES C Fortran)
include(GNUInstallDirs)
find_path(LICENSEBOX_INCLUDE_DIR licensebox_json.h HINTS "$ENV{LICENSEBOX_LIB_DIR}/../include")
find_library(LICENSEBOX_LIBRARY licensebox HINTS "$ENV{LICENSEBOX_LIB_DIR}" REQUIRED)
find_package(Threads REQUIRED)
add_library(licensebox_fortran src/licensebox.f90 src/owner.c)
set_target_properties(licensebox_fortran PROPERTIES Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")
target_include_directories(licensebox_fortran PRIVATE "${LICENSEBOX_INCLUDE_DIR}" PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/modules> $<INSTALL_INTERFACE:include>)
target_link_libraries(licensebox_fortran PUBLIC "${LICENSEBOX_LIBRARY}" Threads::Threads)
add_executable(binding tests/binding.f90)
target_link_libraries(binding PRIVATE licensebox_fortran)
add_executable(adapter tests/adapter.f90)
target_link_libraries(adapter PRIVATE licensebox_fortran)
enable_testing()
add_test(NAME binding COMMAND binding)
install(TARGETS licensebox_fortran EXPORT licenseboxFortranTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules/" DESTINATION include)
install(FILES src/licensebox.f90 DESTINATION share/licensebox-fortran)
install(EXPORT licenseboxFortranTargets DESTINATION lib/cmake/licenseboxFortran)
